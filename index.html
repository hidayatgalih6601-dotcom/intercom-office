PERBAIKAN: Biarkan mikrofon aktif di awal agar SDP WebRTC merekam track audio yang valid */ async function getMic(){ if(!navigator.mediaDevices?.getUserMedia) throw new Error("Browser tidak mendukung akses mikrofon."); if(!localStream){ localStream = await navigator.mediaDevices.getUserMedia({ audio: { echoCancellation: true, noiseSuppression: true, autoGainControl: true }, video: false }); // KUNCI PERBAIKAN: Set ke true agar WebRTC mendeteksi payload stream audio localStream.getAudioTracks().forEach(t => t.enabled = true); } return localStream; } /* 2. PERBAIKAN: Buka blokir Autoplay Browser */ function remoteAudio(pc, id){ pc.ontrack = e => { console.log("REMOTE TRACK", id, e.track.kind, e.streams); diag("REMOTE TRACK", id, e.track.kind); const a = ensureAudioElement(id); if (e.streams && e.streams[0]) { a.srcObject = e.streams[0]; } else if (!a.srcObject) { const ms = new MediaStream(); ms.addTrack(e.track); a.srcObject = ms; } a.muted = false; a.volume = 1; a.autoplay = true; a.playsInline = true; // Paksa play & tangani pemblokiran audio oleh browser a.play().then(() => { console.log("REMOTE AUDIO PLAYING", id); }).catch(err => { console.warn("REMOTE AUDIO PLAY BLOCKED", err); if($("connection")) $("connection").textContent = "⚠ Klik di mana saja pada layar untuk mengaktifkan suara"; }); }; } /* 3. PERBAIKAN: Tekan PTT = Unmute Track */ async function startTalking(){ if(talking || !channel) return; const ok = await acquireSpeaker(); if(!ok){ beep(400); return; } try { const stream = await getMic(); const track = stream.getAudioTracks()[0]; if(!track) throw new Error("Track mikrofon tidak ditemukan."); // Unmute track lokal track.enabled = true; talking = true; beep(1200); $("ptt").classList.add("talking"); $("ptt").classList.remove("locked"); $("pttText").textContent = "SEDANG BICARA..."; $("speakerStatus").textContent = "● KAMU SEDANG BICARA"; $("speakerStatus").className = "text-sm text-red-400 mt-1"; clearInterval(speakerTimer); speakerTimer = setInterval(() => { speakerRef.transaction(v => v && v.uid === me.uid ? {...v, startedAt: firebase.database.ServerValue.TIMESTAMP} : v); }, 5000); } catch(e) { console.error("PTT:", e); await speakerRef.transaction(v => v && v.uid === me.uid ? null : v); alert("Mikrofon tidak tersedia: " + e.message); } } /* 4. PERBAIKAN: Lepas PTT = Mute Track */ async function stopTalking(){ if(!talking) return; // Mute track lokal tanpa memutus koneksi WebRTC if(localStream){ localStream.getAudioTracks().forEach(t => t.enabled = false); } talking = false; clearInterval(speakerTimer); beep(600); $("ptt").classList.remove("talking"); $("pttText").textContent = "TEKAN & TAHAN"; await speakerRef.transaction(v => v && v.uid === me.uid ? null : v); updateSpeaker(await speakerRef.once("value").then(s => s.val())); }
